\section{Experimental Reproducibility Protocol and Additional Tables}
\label{sec:sm-exp}
% ============================================================================

All numerical tables and figures in Section~6 of \texttt{main.pdf} are generated deterministically from fixed NumPy \texttt{default\_rng} seeds using three standalone Python scripts under \texttt{experiments/} (requiring only \texttt{numpy}, \texttt{scipy}, and \texttt{sympy} as listed in \texttt{experiments/requirements.txt}):
{\small
\begin{verbatim}
cd experiments
python3 -m venv .venv && source .venv/bin/activate
pip install -r requirements.txt
python paper_experiments.py      # -> results/paper_experiments.json (Exp. A-D)
python geom_decay_comparison.py  # -> results/geom_decay_comparison.json (Exp. D)
python adversarial_tests.py      # -> results/adversarial_results.json (audit)
\end{verbatim}
}

\paragraph{Exact model parameters, data generation, and seeds.}
In all experiments, covariates $a_1, \dots, a_m$ are drawn i.i.d.\ from $\mathcal{N}(0, I_d)$ and clean responses satisfy $y_i^{\mathrm{clean}} = \max_{1 \le j \le r} \ip{a_i}{\beta_j^*}$:
\begin{itemize}
  \item \textbf{Experiment~A} (dimension and oversampling sweep; \nolinkurl{paper_experiments.py}, seed \texttt{20261120}): $r = 3$ regular-simplex rows in $\R^d$ with mutual pairwise distance $\Delta_{\min} = 1$, $d \in \{5, 10, 20, 40\}$, $m/d \in \{8, 16, 32, 64\}$, $10$ independent Gaussian draws per $(d, m/d)$ pair, worst-case empirical ratio $\hat\mu$ sampled over isotropic, single-block, and uniform-translation perturbations within $\Fnorm{\Delta} \le 0.2\Delta_{\min}$.
  \item \textbf{Experiment~B} (outlier fraction sweep; \nolinkurl{paper_experiments.py}, seed \texttt{20261121}): $d = 10, r = 3, m = 500$, $12$ trials per $\alpha \in \{0.00, 0.10, 0.20, 0.30, 0.40, 0.45, 0.48, 0.50, 0.52, 0.55\}$, comparing (i)~oblivious $+1000$ corruptions at radius $0.15$, (ii)~oblivious Proposition~5.5 corruptions $\xi_i = \ip{a_i}{e_1}$, and (iii)~an omniscient-location adversary selecting the $\alpha m$ rows with largest $|\ip{a_i}{e_1}|$.
  \item \textbf{Experiment~C} ($\Delta_{\min}$-scaling, $r = 2$ scale-free check, and $\kappa$-independence; seed \texttt{20261122}, \nolinkurl{paper_experiments.py}): $d = 10$ with $m = 2000$ for $r = 2$ ($\Delta_{\min} \in \{0.01, 0.05, 0.20, 1.00, 5.00\}$, perturbations up to radius $2.0$) and $m = 600$ for $r = 3$ ($\Delta_{\min}$ sweep at radius $0.15\Delta_{\min}$, plus a $\kappa \in [1.12, 100.00]$ sweep with $\pi_{\min} \ge 0.250$).
  \item \textbf{Experiment~D} (local subgradient diagnostics and analytical cone curves; \nolinkurl{paper_experiments.py} seed \texttt{20261123} and \nolinkurl{geom_decay_comparison.py} seeds \texttt{20261123--20261127}): $d = 20, r = 3, m = 500$, regular-simplex $\beta^*$ ($\Delta_{\min} = 1$), $\pm 100$ oblivious corruptions on the first $\alpha m$ rows ($\alpha \in \{0.00, 0.20, 0.30, 0.40\}$), $250$ iterations from the same local initialization $\beta^{(0)} = \beta^* + D_0$ with $\Fnorm{D_0} = 0.25$. \texttt{PolyakSGM} uses oracle step size $(f(\beta^{(k)}) - f(\beta^*))/\Fnorm{g^{(k)}}^2$ (stopping early when $f(\beta^{(k)}) - f(\beta^*) \le 10^{-14}$), whereas \texttt{GeomDecaySGM} uses no knowledge of $f(\beta^*)$ or $f^*$, updating $\beta^{(k+1)} = \beta^{(k)} - \lambda_0 q^k g^{(k)}/\Fnorm{g^{(k)}}$ with a single fixed schedule $(\lambda_0, q) = (0.04, 0.95)$ across all $\alpha$ and all seeds.
\end{itemize}

\paragraph{Explicit omniscient-location breakdown at $\alpha = 0.25$.}
In \nolinkurl{experiments/adversarial_tests.py} (seed~\texttt{4040}), for a Gaussian draw $A \in \R^{m \times d}$, selecting the $\alpha m$ rows with largest $|\ip{a_i}{e_1}|$ (Model~(iv), omniscient support) at $\alpha = 0.25$ and setting $\xi_i = \ip{a_i}{e_1}$ yields $\sum_{i \in S_{\mathrm{bad}}} |\ip{a_i}{e_1}| > \sum_{i \notin S_{\mathrm{bad}}} |\ip{a_i}{e_1}|$, because the top $25\%$ tail of $|Z|$ ($Z \sim \mathcal{N}(0, 1)$, threshold $z_{0.25} = \Phi^{-1}(0.875) \approx 1.1503$) carries fraction
\[
  \frac{\E[|Z|\,\ind_{\{|Z| \ge z_{0.25}\}}]}{\E[|Z|]} = e^{-z_{0.25}^2 / 2} \approx 0.516 > \frac{1}{2}
\]
of the total $\ell_1$ mass. Hence uniform translations $\tilde\beta(t) = \beta^* + t e_1$ strictly decrease $f$ for all $t \in (0, 1]$ (in the seed-\texttt{4040} instance, $f(\tilde\beta(t)) - f(\beta^*) = -0.011841\,t < 0$), breaking local optimality of $\beta^*$ at $\alpha = 0.25$ and confirming that oblivious support selection in Theorem~5.1 is essential for the $\alpha < 1/2$ threshold.
